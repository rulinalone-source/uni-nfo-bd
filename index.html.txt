            /* 💡 উপরে দেওয়া নিয়মে আপনি চাইলে আরও নতুন ভার্সিটির কোড ব্লক যোগ করতে পারেন */

        // ⚙️ ওয়েবসাইট ফিল্টারিং ও রেন্ডারিং ফাংশনসমূহ (এখানে এডিট করার প্রয়োজন নেই)

        // ১. কার্ড রেন্ডার করা (হোম পেইজে ভার্সিটির লিস্ট দেখাবে)
        function renderCards(dataList) {
            const grid = document.getElementById('university-grid');
            const noResults = document.getElementById('no-results');
            const resultCount = document.getElementById('result-count');
            
            grid.innerHTML = '';
            resultCount.innerText = `মোট প্রাপ্ত বিশ্ববিদ্যালয়: ${dataList.length}টি`;

            if (dataList.length === 0) {
                noResults.classList.remove('hidden');
            } else {
                noResults.classList.add('hidden');
                dataList.forEach(uni => {
                    const card = document.createElement('div');
                    card.className = "bg-white rounded-lg shadow-md overflow-hidden hover:shadow-xl transition cursor-pointer transform hover:-translate-y-1 flex flex-col justify-between";
                    card.onclick = () => showDetailsPage(uni.id);

                    card.innerHTML = `
                        <div>
                            <img src="${uni.image}" alt="${uni.name}" class="w-full h-40 object-cover">
                            <div class="p-4">
                                <h3 class="text-lg font-bold text-gray-800 leading-snug mb-1">${uni.name}</h3>
                                <span class="inline-block bg-blue-100 text-blue-800 text-xs font-semibold px-2 py-0.5 rounded mb-2">${uni.category}</span>
                                <p class="text-gray-500 text-xs mb-2">📍 ${uni.location} (${uni.division} বিভাগ)</p>
                                <p class="text-gray-600 text-xs line-clamp-2">${uni.description || ''}</p>
                            </div>
                        </div>
                        <div class="p-4 pt-0">
                            <button class="w-full bg-blue-600 hover:bg-blue-700 text-white text-xs font-semibold py-2 rounded transition">
                                বিস্তারিত দেখুন →
                            </button>
                        </div>
                    `;
                    grid.appendChild(card);
                });
            }
        }

        // ২. ফিল্টারিং লজিক (স্মার্ট সার্চ, বিভাগ ফিল্টার, গ্রুপ ফিল্টার)
        function applyFilters() {
            const searchVal = document.getElementById('search-input').value.toLowerCase().trim();
            const streamVal = document.getElementById('stream-filter').value;
            const categoryVal = document.getElementById('category-filter').value;
            const locationVal = document.getElementById('location-filter').value;

            const filtered = universities.filter(uni => {
                // গ্রুপ বা শাখা ম্যাচিং
                const matchStream = (streamVal === 'all') || (uni.streams && uni.streams.includes(streamVal));
                
                // ক্যাটাগরি ম্যাচিং
                const matchCategory = (categoryVal === 'all') || (uni.category === categoryVal);
                
                // লোকেশন / বিভাগ ম্যাচিং (সরাসরি টেক্সট মিলানো)
                const matchLocation = (locationVal === 'all') || (uni.division && uni.division.trim() === locationVal.trim());

                // স্মার্ট সার্চ ম্যাচিং (কাছাকাছি নাম, অ্যাক্রোনিম বা কীওয়ার্ড দিয়ে সার্চ)
                let matchSearch = false;
                if (!searchVal) {
                    matchSearch = true;
                } else {
                    const searchWords = searchVal.split(" ");
                    matchSearch = searchWords.every(word => {
                        return uni.name.toLowerCase().includes(word) ||
                               (uni.acronym && uni.acronym.toLowerCase().includes(word)) ||
                               (uni.location && uni.location.toLowerCase().includes(word)) ||
                               (uni.keywords && uni.keywords.some(k => k.toLowerCase().includes(word)));
                    });
                }

                return matchStream && matchCategory && matchLocation && matchSearch;
            });

            renderCards(filtered);
        }

        // ৩. নির্দিষ্ট ভার্সিটির বিস্তারিত পেইজ দেখানো
        function showDetailsPage(uniId) {
            const uni = universities.find(u => u.id === uniId);
            if (!uni) return;

            document.getElementById('uni-name').innerText = uni.name;
            document.getElementById('uni-category').innerText = uni.category;
            document.getElementById('uni-location').innerText = "📍 " + uni.location + " (" + uni.division + " বিভাগ)";
            document.getElementById('uni-image').src = uni.image;
            document.getElementById('uni-description').innerText = uni.description || "তথ্য উপলব্ধ নয়।";
            document.getElementById('uni-requirements').innerText = uni.requirements || "তথ্য উপলব্ধ নয়।";

            const subjectsList = document.getElementById('uni-subjects');
            subjectsList.innerHTML = '';
            if(uni.subjects && uni.subjects.length > 0) {
                uni.subjects.forEach(subject => {
                    const li = document.createElement('li');
                    li.innerText = subject;
                    subjectsList.appendChild(li);
                });
            } else {
                subjectsList.innerHTML = '<li>কোনো বিষয় যুক্ত করা হয়নি।</li>';
            }

            document.getElementById('home-page').classList.add('hidden');
            document.getElementById('details-page').classList.remove('hidden');
            window.scrollTo(0, 0);
        }

        // ৪. হোম পেইজে ফিরে যাওয়ার ফাংশন
        function showHomePage() {
            document.getElementById('details-page').classList.add('hidden');
            document.getElementById('home-page').classList.remove('hidden');
            applyFilters();
            window.scrollTo(0, 0);
        }

        // প্রথমবার পেজ লোড হলে ফিল্টার ও রেন্ডার চালানো
        applyFilters();
    </script>
